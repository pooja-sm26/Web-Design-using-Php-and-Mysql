-- ============================================================
-- TRAFFIC ROUTE MAP DATABASE SCHEMA
-- Compatible with MySQL 5.7+ / MariaDB 10.3+
-- ============================================================

CREATE DATABASE IF NOT EXISTS traffic_map CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE traffic_map;

-- ============================================================
-- TABLE: locations
-- Points of interest, waypoints, intersections
-- ============================================================
CREATE TABLE IF NOT EXISTS locations (
    id           INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name         VARCHAR(150)   NOT NULL,
    category     ENUM('city','district','landmark','intersection','transit','hospital','school','commercial','residential','industrial') NOT NULL DEFAULT 'landmark',
    lat          DECIMAL(10,7)  NOT NULL,
    lng          DECIMAL(10,7)  NOT NULL,
    address      VARCHAR(255),
    description  TEXT,
    is_active    TINYINT(1)     NOT NULL DEFAULT 1,
    created_at   TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at   TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_lat_lng (lat, lng),
    INDEX idx_category (category),
    INDEX idx_name (name)
) ENGINE=InnoDB;

-- ============================================================
-- TABLE: routes
-- Named routes connecting two locations with optional waypoints
-- ============================================================
CREATE TABLE IF NOT EXISTS routes (
    id              INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name            VARCHAR(200)   NOT NULL,
    origin_id       INT UNSIGNED   NOT NULL,
    destination_id  INT UNSIGNED   NOT NULL,
    distance_km     DECIMAL(8,3)   NOT NULL DEFAULT 0,
    duration_min    SMALLINT UNSIGNED NOT NULL DEFAULT 0,
    route_type      ENUM('highway','arterial','local','expressway','rural') NOT NULL DEFAULT 'arterial',
    polyline        MEDIUMTEXT     COMMENT 'Encoded polyline string or JSON array of [lat,lng] pairs',
    toll            TINYINT(1)     NOT NULL DEFAULT 0,
    is_active       TINYINT(1)     NOT NULL DEFAULT 1,
    created_at      TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (origin_id)      REFERENCES locations(id) ON DELETE RESTRICT,
    FOREIGN KEY (destination_id) REFERENCES locations(id) ON DELETE RESTRICT,
    INDEX idx_origin (origin_id),
    INDEX idx_destination (destination_id),
    INDEX idx_route_type (route_type)
) ENGINE=InnoDB;

-- ============================================================
-- TABLE: traffic_segments
-- Real-time or periodic traffic status per road segment
-- ============================================================
CREATE TABLE IF NOT EXISTS traffic_segments (
    id              INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    route_id        INT UNSIGNED   NOT NULL,
    segment_index   SMALLINT UNSIGNED NOT NULL DEFAULT 0  COMMENT 'Order within route',
    start_lat       DECIMAL(10,7)  NOT NULL,
    start_lng       DECIMAL(10,7)  NOT NULL,
    end_lat         DECIMAL(10,7)  NOT NULL,
    end_lng         DECIMAL(10,7)  NOT NULL,
    speed_kmh       TINYINT UNSIGNED NOT NULL DEFAULT 60,
    congestion      ENUM('free','light','moderate','heavy','standstill') NOT NULL DEFAULT 'free',
    incident        TINYINT(1)     NOT NULL DEFAULT 0,
    recorded_at     TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (route_id) REFERENCES routes(id) ON DELETE CASCADE,
    INDEX idx_route_segment (route_id, segment_index),
    INDEX idx_congestion (congestion),
    INDEX idx_recorded_at (recorded_at)
) ENGINE=InnoDB;

-- ============================================================
-- TABLE: incidents
-- Accidents, roadworks, closures, events
-- ============================================================
CREATE TABLE IF NOT EXISTS incidents (
    id           INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    route_id     INT UNSIGNED   NULL  COMMENT 'NULL = point incident not tied to a route',
    type         ENUM('accident','roadwork','closure','flood','event','hazard','police') NOT NULL,
    severity     ENUM('low','medium','high','critical') NOT NULL DEFAULT 'medium',
    title        VARCHAR(200)   NOT NULL,
    description  TEXT,
    lat          DECIMAL(10,7)  NOT NULL,
    lng          DECIMAL(10,7)  NOT NULL,
    reported_by  VARCHAR(100),
    starts_at    DATETIME       NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ends_at      DATETIME,
    is_active    TINYINT(1)     NOT NULL DEFAULT 1,
    created_at   TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (route_id) REFERENCES routes(id) ON DELETE SET NULL,
    INDEX idx_type (type),
    INDEX idx_severity (severity),
    INDEX idx_active (is_active),
    INDEX idx_lat_lng (lat, lng)
) ENGINE=InnoDB;

-- ============================================================
-- TABLE: searches
-- User search history for analytics / autocomplete
-- ============================================================
CREATE TABLE IF NOT EXISTS searches (
    id           INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    session_id   VARCHAR(64),
    origin       VARCHAR(200),
    destination  VARCHAR(200),
    mode         ENUM('driving','walking','transit','cycling') NOT NULL DEFAULT 'driving',
    route_id     INT UNSIGNED   NULL,
    searched_at  TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (route_id) REFERENCES routes(id) ON DELETE SET NULL,
    INDEX idx_session (session_id),
    INDEX idx_searched_at (searched_at)
) ENGINE=InnoDB;

-- ============================================================
-- TABLE: saved_routes
-- User bookmarked / favourite routes
-- ============================================================
CREATE TABLE IF NOT EXISTS saved_routes (
    id           INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    session_id   VARCHAR(64)    NOT NULL,
    label        VARCHAR(100)   NOT NULL DEFAULT 'My Route',
    route_id     INT UNSIGNED   NOT NULL,
    notes        TEXT,
    saved_at     TIMESTAMP      NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (route_id) REFERENCES routes(id) ON DELETE CASCADE,
    INDEX idx_session (session_id)
) ENGINE=InnoDB;

-- ============================================================
-- TABLE: map_layers
-- Toggle-able overlays (hospitals, schools, transit stops …)
-- ============================================================
CREATE TABLE IF NOT EXISTS map_layers (
    id           INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    name         VARCHAR(100)   NOT NULL,
    slug         VARCHAR(60)    NOT NULL UNIQUE,
    icon         VARCHAR(10)    NOT NULL DEFAULT '📍',
    color        VARCHAR(7)     NOT NULL DEFAULT '#3B82F6',
    is_default   TINYINT(1)     NOT NULL DEFAULT 0
) ENGINE=InnoDB;

-- ─────────────────────────────────────────────────────────────────────────
-- CART  (one row per user+medicine; quantity increments on duplicate add)
-- ─────────────────────────────────────────────────────────────────────────
CREATE TABLE cart (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    user_id     INT NOT NULL,
    medicine_id INT NOT NULL,
    quantity    INT NOT NULL DEFAULT 1,
    added_at    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY uniq_user_medicine (user_id, medicine_id),
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (medicine_id) REFERENCES medicines(id) ON DELETE CASCADE
) ENGINE=InnoDB;

-- ─────────────────────────────────────────────────────────────────────────
-- ORDERS & ORDER ITEMS
-- ─────────────────────────────────────────────────────────────────────────
CREATE TABLE orders (
    id                INT AUTO_INCREMENT PRIMARY KEY,
    user_id           INT NOT NULL,
    total_amount      DECIMAL(10,2) NOT NULL DEFAULT 0,
    shipping_address  TEXT NOT NULL,
    status            ENUM('Pending','Processing','Shipped','Delivered','Cancelled') NOT NULL DEFAULT 'Pending',
    order_date        TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB;

CREATE TABLE order_items (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    order_id    INT NOT NULL,
    medicine_id INT NOT NULL,
    quantity    INT NOT NULL DEFAULT 1,
    price       DECIMAL(10,2) NOT NULL DEFAULT 0,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
    FOREIGN KEY (medicine_id) REFERENCES medicines(id) ON DELETE CASCADE
) ENGINE=InnoDB;
